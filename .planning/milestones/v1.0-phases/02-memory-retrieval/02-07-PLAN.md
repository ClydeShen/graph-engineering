  Dedup hash computation:
  ```typescript
  function computeDedupHash(scopeId: string, entityId: string, eventType: string, content: string): string {
    const payloadHash = createHash('sha256').update(content).digest('hex');
    return createHash('sha256')
      .update(`${scopeId}|${entityId}|${eventType}|${payloadHash}`)
      .digest('hex');
  }
  ```

  Export:
  ```typescript
  export async function insertWorkingMemory(
    pool: Pool,
    scopeId: string,
    entityId: string,
    eventType: string,
    content: string,
  ): Promise<{ inserted: boolean }> {
    const dedupHash = computeDedupHash(scopeId, entityId, eventType, content);

    // 5-minute dedup window: check for identical hash within last 5 minutes
    const existing = await pool.query<{ id: string }>(
      `SELECT id FROM working_memory
       WHERE scope_id = $1
         AND dedup_hash = $2
         AND created_at > NOW() - INTERVAL '5 minutes'
       LIMIT 1`,
      [scopeId, dedupHash],
    );
    if (existing.rows.length > 0) {
      return { inserted: false };
    }

    await pool.query(
      `INSERT INTO working_memory (scope_id, content, dedup_hash, created_at)
       VALUES ($1, $2, $3, NOW())`,
      [scopeId, writeGuard(content), dedupHash],
    );
    return { inserted: true };
  }
  ```

  Notes:
  - Hash is computed from raw content (before writeGuard) — consistent dedup across identical raw content

  - entity_id is not a column in working_memory (it's in the hash but not stored separately) — check migration 003 to confirm
  - ts_doc is GENERATED ALWAYS — do NOT pass it in INSERT

  ```typescript
  export class ConflictResolverWorker {
    private readonly llm: LLMProvider;

    constructor(llm: LLMProvider) {
      this.llm = llm;
    }

    async onConflict(
      entityId: string,
      payloadA: string,
      payloadB: string,
    ): Promise<{ merged: string } | { skipped: true }> {
      // Mutex guard — prevent concurrent LLM merge on same entity
      if (ActiveResolverRegistry.get(entityId)) {
        return { skipped: true };
      }

      ActiveResolverRegistry.set(entityId, true);
      try {
        // LLM CALL — ADR 22 (semantic merge; cannot be deterministic)
        const merged = await this.llm.chat([
          { role: 'system', content: 'You are a conflict resolver. Given two conflicting versions of an execution payload, produce a single merged version that preserves the intent of both.' },
          { role: 'user', content: `Version A:\n${writeGuard(payloadA)}\n\nVersion B:\n${writeGuard(payloadB)}` },
        ]);
        return { merged };
      } finally {
        ActiveResolverRegistry.delete(entityId);
      }
    }
  }
  ```
