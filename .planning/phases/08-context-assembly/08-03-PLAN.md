<objective>
Implement D-03/D-04/D-05/D-12 (CCR reversible compression) by creating `packages/workers/src/context/ccr.ts`, wiring CCR sentinel injection + dropped-event store into `assembleContext()`, and adding a `runContextAssemblyPipeline()` orchestration helper that calls the Plan 02 pipeline hooks (D-06/D-07/D-08/D-13) with a populated `PipelineContext`.

Purpose: This is the CCR Level-3-degradation replacement (ADR-13 supplement) — instead of hard-suspending a Scope when the causal chain exceeds W_max, dropped events are summarized into a `<<ccr:HASH N_dropped>>` sentinel, their payloads are cached in an invocation-scoped Map, and the LLM is given both a `memex_retrieve` tool (structural) and system-prompt-adjacent instructions (directional) to fetch them back on demand. This plan also resolves the two open questions from 08-PATTERNS.md: (1) `memex_retrieve` routing lives in `ccr.ts` as a `Tool`-shaped `execute()` handler closing over the per-invocation `CcrStore`, returned alongside the tool definition for the calling Worker to register; (2) `PipelineContext` is consumed here (defined in Plan 02's `worker.abstract.ts`) — no separate `pipeline-context.ts` file is created. It also resolves the D-03/ADR-30-D-1 tension: CCR instructions are a NEW field on `AssembledContext` (`ccrInstructions?: string`), kept entirely separate from `STABLE_SYSTEM_ROLE`, which remains byte-identical across invocations for Anthropic prompt-cache eligibility.

Output: `packages/workers/src/context/ccr.ts` (new), `packages/workers/src/context/assemble.ts` (extended), with unit tests.

```typescript
import type { EventLogNode } from '@shared/types';
import { countTokens } from '@shared/tokenizer';
import { knapsackSlice, type KnapsackGraph } from './knapsack.js';

export interface AssembledContext {
  stable: string;
  context: EventLogNode[] | null;
  volatile: string;
}

export const STABLE_SYSTEM_ROLE =
  'You are a graph-native agent operating on an append-only Execution Graph. ' +
  'Your context window is a read-time projection of the graph state. ' +
  'All persistent writes occur through the GraphHandle write interface only. ' +
  'Context overflow is handled by the sliding-window discarder — older events ' +
  'are dropped, not summarized. Retrieve older context via graph queries if needed.';

export function computeContextBudgets(params: {
  wMax: number;
  stableTokens: number;
  volatileTokens: number;
}): { forKnapsack: number } {
  return {
    forKnapsack: Math.max(0, params.wMax - params.stableTokens - params.volatileTokens),
  };
}

export async function assembleContext(
  graph: KnapsackGraph,
  scopeId: string,
  rootHash: string,
  currentInput: unknown,
  wMax: number,
  scopeClosed = false
): Promise<AssembledContext> {
  const stable = STABLE_SYSTEM_ROLE;
  const volatile = JSON.stringify(currentInput);

  if (scopeClosed) {
    return { stable, context: null, volatile };
  }

  const stableTokens = countTokens(stable);
  const volatileTokens = countTokens(volatile);
  const { forKnapsack: contextBudget } = computeContextBudgets({ wMax, stableTokens, volatileTokens });

  // dropped is available for Phase 08 CCR marker injection; ignored here (ADR 13 supplement)
  const contextEvents = (await knapsackSlice(graph, scopeId, rootHash, contextBudget)).kept;

  return { stable, context: contextEvents, volatile };
}
```
