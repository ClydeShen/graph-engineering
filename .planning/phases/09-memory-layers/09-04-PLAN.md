    IMPORTS:
    import type { Pool } from 'pg'
    import type { EmbeddingProvider } from '@graph/shared'
    import { countTokens } from '@shared/tokenizer'

    TYPES — export:
    MemReflectInput: { query_text: string; trigger_type: 'cold_start' | 'conflict_detected' | 'macro_planning'; w_max: number; scope_id: string }
    MemReflectOutput: { content: string; tokens: number; sections: { procedural: string; episodic: string; semantic: string } }

    BUDGET HELPER — export function computeReflectBudget(triggerType: string, wMax: number): number:
    if triggerType === 'conflict_detected' return Math.min(1000, Math.floor(wMax * 0.2))
    else return Math.min(2000, Math.floor(wMax * 0.3))

    PROC LIMIT HELPER — private function procLimit(triggerType: string): number:
    return triggerType === 'conflict_detected' ? 1 : 3

    HYBRID SEARCH FUNCTION — private async hybridSearchEpisodic(pool, queryEmbeddingLiteral, queryText, limit):
    Runs the RRF CTE against episodic_memory. SELECT columns: id, intent_summary, outcome_summary.
    Vector path: ORDER BY embedding <=> $1::vector LIMIT 20 (where $1 = queryEmbeddingLiteral)
    BM25 path: plainto_tsquery('english', $2) (where $2 = queryText) WHERE ts_doc @@ query LIMIT 20
    RRF: 0.6*(1.0/(60+COALESCE(vc.vector_rank,21))) + 0.4*(1.0/(60+COALESCE(bc.bm25_rank,21))) AS rrf_score
    JOIN back to episodic_memory e ON r.id = e.id
    ORDER BY rrf_score DESC LIMIT $3 (= limit param)
    Return rows as Array<{ id: string; intent_summary: string | null; outcome_summary: string | null; rrf_score: number }>

    HYBRID SEARCH FUNCTION — private async hybridSearchSemantic(pool, queryEmbeddingLiteral, queryText, limit):
    Same RRF CTE against semantic_memory. Add WHERE superseded_by IS NULL in both vector and BM25
    sub-queries. SELECT columns: id, content, rrf_score.
    Return rows as Array<{ id: string; content: string; rrf_score: number }>

    HYBRID SEARCH FUNCTION — private async hybridSearchProcedural(pool, queryEmbeddingLiteral, queryText, limit):
    Same RRF CTE against procedural_memory. Use intent_embedding column for vector path (not topology_embedding
    — intent_embedding is 1536-dim text embedding). Add WHERE is_anti_pattern = FALSE in both sub-queries.
    SELECT columns: id, intent_description, template_graph, rrf_score.
    Return rows as Array<{ id: string; intent_description: string | null; template_graph: unknown; rrf_score: number }>

    FORMAT HELPERS:
    formatProcedural(rows, budgetTokens): iterates rows descending by rrf_score, for each row
    builds string "## Procedural Pattern\nIntent: {intent_description}\n".
    If countTokens(singleEntry) > budgetTokens * 0.6, emit only intent_description (skip template_graph).
    Else emit full JSON. Accumulate until budget exhausted. Return formatted string.

    formatEpisodic(rows, budgetTokens): iterates rows, builds "- {intent_summary}: {outcome_summary}"
    per row, stops when accumulated tokens exceed budgetTokens. Return formatted string or empty string.

    formatSemantic(rows, budgetTokens): iterates rows, builds "- {content}" per row, stops when
    accumulated tokens exceed budgetTokens. Return formatted string or empty string.

    MAIN EXPORT — export async function memReflect(pool: Pool, embed: EmbeddingProvider, input: MemReflectInput): Promise<MemReflectOutput>:

    const budget = computeReflectBudget(input.trigger_type, input.w_max)
    const limit = procLimit(input.trigger_type)
    // LLM CALL — ADR 22 (embedding for query text; not counted against Worker token budget)
    const { vector } = await embed.embed(input.query_text)
    const queryEmbeddingLiteral = '[' + vector.join(',') + ']'

    Step 1 — Procedural:
    const procRows = await hybridSearchProcedural(pool, queryEmbeddingLiteral, input.query_text, limit)
    const procText = formatProcedural(procRows, budget)
    const pTokens = countTokens(procText)

    Step 2 — Episodic:
    const epiRows = await hybridSearchEpisodic(pool, queryEmbeddingLiteral, input.query_text, 5)
    const epiText = formatEpisodic(epiRows, Math.max(0, budget - pTokens))
    const eTokens = countTokens(epiText)

    Step 3 — Semantic:
    const semRows = await hybridSearchSemantic(pool, queryEmbeddingLiteral, input.query_text, 5)
    const semText = formatSemantic(semRows, Math.max(0, budget - pTokens - eTokens))
    const sTokens = countTokens(semText)

    Assemble [REFLECTION MEMORY] content:
    Build content string with non-empty sections only:
    "## Procedural Memory\n{procText}\n\n## Episodic Memory\n{epiText}\n\n## Semantic Memory\n{semText}"
    (skip section header + content if the section text is empty string)

    Return { content, tokens: pTokens + eTokens + sTokens, sections: { procedural: procText, episodic: epiText, semantic: semText } }
  </action>
